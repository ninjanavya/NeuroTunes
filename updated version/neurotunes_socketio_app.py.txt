    function onYouTubeIframeAPIReady() {
      player = new YT.Player('player', {
        height: '340', width: '100%', videoId: '',
        playerVars: { playsinline: 1, rel: 0, modestbranding: 1 },
        events: {
          onReady: function() {
            playerReady = true;
            if (pendingVideoId) {
              player.loadVideoById(pendingVideoId);
              pendingVideoId = null;
            }
          },
          onStateChange: function(event) {
            if (!currentJamCode || suppressPlayerBroadcast) return;
            if (event.data === YT.PlayerState.PLAYING) emitPlayback('playing');
            if (event.data === YT.PlayerState.PAUSED) emitPlayback('paused');
            if (event.data === YT.PlayerState.ENDED) {
              socket.emit('next_song', { roomCode: currentJamCode, username: currentUserName });
            }
          }
        }
      });
    }

    function getUserName() {
      const name = document.getElementById('friendNameInput').value.trim();
      currentUserName = name || 'Guest';
      return currentUserName;
    }

    function setStatus(id, cls, msg) {
      const el = document.getElementById(id);

    function playVideoById(videoId) {
      if (!videoId) return;
      if (playerReady && player && typeof player.loadVideoById === 'function') {
        player.loadVideoById(videoId);
        setTimeout(() => { try { player.playVideo(); } catch (e) {} }, 250);
      } else {
        pendingVideoId = videoId;
      }
    }

    }
